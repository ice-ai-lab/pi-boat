迁移脚本是幂等的：重复执行不会二次改写，断言失败时会整体回滚，不会留下半成品数据。

核心映射逻辑：

typescript 
const LEGACY_DIRECTIONS = { 'D': 'debit', 'C': 'credit' } as const;

export function normalize(types: AccountType[]): Normalized[] {
  // 幂等：已经是三段式的条目直接跳过
  return types.map((t) => {
    if (t.account && t.fund && t.direction) return t;
    return {
      account: pickAccount(t.code),
      fund: pickFund(t.code),
      direction: LEGACY_DIRECTIONS[t.direction] ?? t.direction,
    };
  });
}
